% 浮动体与正文的垂直间距。
% \intextsep 上下相同，但 TeX 只在下方补完整基线胶水（约12pt，上方仅

% ===================== 与学校下发的 Word 版开题报告模板对齐 =====================
% 说明：hithesisart.cls 是按「研究生学位论文」规范写的（版心 150mm、行距 20.5bp），
% 而学校下发的本科开题报告 Word 模板是上下左右各 2.5cm、版心 160mm×247mm、正文
% 1.5 倍行距（宋体小四对应 22pt）。以下四项把 LaTeX 版对齐到 Word 模板。
\makeatletter

% (1) 删除封面隐形水印
%     hithesisart.cls:546 会在本科封面顶部插入一行白色（不可见）文字「嗨！thesis」。
%     屏幕/打印都看不见，但复制文字、查重系统、文本提取会把它读出来。
%     这里换成一个等高的透明占位，既清掉文字，又保持封面垂直排版与原来完全一致。
%     若不在意封面整体上移约 16pt，也可直接写 \def\hit@hi{}。
\def\hit@hi{\vphantom{X}}%

% (2) 正文行距 20.50398bp → 22bp（等价 Word 1.5 倍行距）
%     只改 \normalsize，四级标题等其它字号的基线距不受影响。
\renewcommand\normalsize{%
  \@setfontsize\normalsize{12bp}{22bp}%
  \abovedisplayskip=8pt
  \abovedisplayshortskip=8pt
  \belowdisplayskip=\abovedisplayskip
  \belowdisplayshortskip=\abovedisplayshortskip
}
\AtBeginDocument{\normalsize}
\makeatother

% (3) 版心与页边距：上下左右各 25mm，版心 160mm×247mm
\geometry{text={160true mm,247true mm},left=25true mm}

% (4) 首行位置补偿：Word 会在首行上方留出整行行距，LaTeX 默认只留 \topskip=10pt，
%     补偿到 23.4pt 后，首行文字落点与 Word 模板一致。
\topskip=23.4pt

% (5) 各级标题的上下间距：对齐 Word 模板。
%     模板实测（标题1/标题2 = 段前段后7.8pt/0pt + 22pt固定行距）：
%     一级标题上下留白各约 14pt，二级标题上约 7pt、下约 8.5pt。
%     cls 原值 section 7.5bp/7.5bp、subsection 7bp/7bp，实际视觉留白只有
%     5pt 左右——这就是“整页都是字、显得密”的主要原因。以下按模板实测值放大。
\ctexset{
  section={beforeskip=15.4bp, afterskip=16.3bp},
  subsection={beforeskip=8.5bp, afterskip=10.45bp},
}
